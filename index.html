function calculerFrais() { const montantInput = document.getElementById('montant').value; const recap = document.getElementById('recap-frais'); const montant = parseFloat(montantInput); const TAUX_COMMISSION = 0.015; if (!isNaN(montant) && montant > 0) { const frais = montant * TAUX_COMMISSION; const total = montant + frais; document.getElementById('frais-valeur').innerText = frais.toLocaleString('fr-FR'); document.getElementById('total-valeur').innerText = total.toLocaleString('fr-FR'); document.getElementById('net-valeur').innerText = montant.toLocaleString('fr-FR'); recap.style.display = 'block'; } else { recap.style.display = 'none'; } }
